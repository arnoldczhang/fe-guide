Micro.tsx
```jsx
import { loadMicroApp, registerMicroApps } from 'qiankun';

function micro() {
  const microRef = useRef<any>(null);
  useEffect(() => {
    loadMicroApp({
      name: 'first', // @important 这里的name和子应用要对应
      entry: '//localhost:5173',
      container: microRef.current,
      props: { brand: 'xxx' },
    });
    return () => null;
  }, []);
  return (
    <article>
      <section ref={microRef}></section>
    </article>
  )
}

export default micro;
```


**vue3改动点**

1. 构建配置适配qiankun
2. 路由适配微前端场景

vite.config.ts

```typescript
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import qiankun from "vite-plugin-qiankun"

// useDevMode 开启时与热更新插件冲突
const useDevMode = true

// https://vite.dev/config/
const baseConfig = defineConfig({
  plugins: [
    vue(),
    qiankun('first', { // @important 微应用名字，与主应用注册的微应用名字保持一致
      useDevMode,
    }),
  ],
  server: {
    host: '0.0.0.0',
    port: 5173,
    cors: true,
  },
})

// @important 线上环境指向真实ip和端口
export default ({ mode }: any) => {
  baseConfig.base = 'http://127.0.0.1:5173/';
  if (mode === 'development') {
    baseConfig.base = '/';
  }
  return baseConfig;
}
```

main.ts

```typescript
import { createApp } from 'vue'
import './style.css'
import MainApp from './App.vue'
import router, { history } from './router'
import {
  renderWithQiankun,
  qiankunWindow
} from 'vite-plugin-qiankun/dist/helper';

let instance: any = null; 
async function render(props: any = {}) {
  const { container, activeRule } = props;
  instance = createApp(MainApp);
  instance.use(router)
  instance.mount(container || '#app');
  instance.config.globalProperties.$activeRule = activeRule;
}

const initQianKun = () => {
  renderWithQiankun({
      mount(props) {
          render(props);
      },
      bootstrap() { },
      update() { },
      unmount() {
          instance.unmount();
          instance._container.innerHTML = "";
          instance = null;
          history.destroy();
      },
  });
};

// @important 区分微前端和单独打开场景
if (qiankunWindow.__POWERED_BY_QIANKUN__) {
  console.log("qiankun渲染开始");
  initQianKun();
} else {
  console.log("不是qiankun渲染");
  render();
}

export default render;
```

router.ts

```typescript
import { createRouter, createWebHistory } from 'vue-router'
import { qiankunWindow } from "vite-plugin-qiankun/dist/helper";

import Home from '../views/Home.vue'

const routes: Array<any> = [
  {
    path: '/',
    name: 'Home',
    component: Home
  },
  {
    path: '/about',
    name: 'About',
    component: () => import('../views/About.vue')
  }
]

// @important 适配基座路由
export const history = createWebHistory(
  qiankunWindow.__POWERED_BY_QIANKUN__ ? '#/micro' : "/"
);

const router = createRouter({
  history,
  routes
})

export default router
```
