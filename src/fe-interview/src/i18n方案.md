```typescript
// vite.config.ts
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import viteAutoI18n, { YoudaoTranslator } from 'vite-auto-i18n-plugin'

// https://vite.dev/config/
const baseConfig = defineConfig({
  plugins: [
    vue(),
    viteAutoI18n({
      targetLangList: ['en'],
      translator: new YoudaoTranslator({
        appId: 'appId', // https://ai.youdao.com/console/#/
        appKey: 'appKey'
      }),
    }),
  ],
})

export default ({ mode }: any) => {
  baseConfig.base = 'http://127.0.0.1:5173/';
  if (mode === 'development') {
    baseConfig.base = '/';
  }
  return baseConfig;
}
```
